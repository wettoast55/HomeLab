#proxmox host installing tailscale, assuming proxmox configured already on device
curl -fsSL https://tailscale.com/install.sh | sh

#bringing it onlinle and setting subnet to advertise (all within range of 192.168.50.1-192.168.50.254)
tailscale up --advertise-routes=192.168.50.0/24

#enable ip forwarding
sysctl net.ipv4.ip_forward

#if doesnt look like net.ipv4.ip_forward = 1, update it
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf
sysctl -p
