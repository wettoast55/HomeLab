#set up of telegram notifications ... ups script
nano /usr/local/bin/ups-telegram.sh

#replace NEW_BOT_TOKEN and actual chat id
"
#!/bin/bash
  TOKEN="NEW_BOT_TOKEN"
  CHAT_ID="8837651042"
  
  MESSAGE="$1"
  
  curl -s -X POST \
  "https://api.telegram.org/bot${TOKEN}/sendMessage" \
  -d chat_id="${CHAT_ID}" \
  -d text="${MESSAGE}" >/dev/null
"

#make script executable and test
chmod +x /usr/local/bin/ups-telegram.sh
/usr/local/bin/ups-telegram.sh "UPS is on battery"
