"
#!/bin/bash

#denotes ups status type/location
UPS="cst1500s"
STATEFILE="/tmp/ups_status"

#for changes in ups
CURRENT=$(upsc "$UPS" ups.status 2>/dev/null)
PREVIOUS=$(cat "$STATEFILE" 2>/dev/null)

#if ups on battery power
if [[ "$CURRENT" == "OB" && "$PREVIOUS" != "OB" ]]; then
    /usr/local/bin/send-telegram.sh "ALERT: UPS ON BATTERY - Proxmox is running on battery power"
fi

#if ups back on line power
if [[ "$CURRENT" == OL* && "$PREVIOUS" == "OB" ]]; then
    /usr/local/bin/send-telegram.sh "ALERT: UPS POWER RESTORED - Utility power has returned"
fi

echo "$CURRENT" > "$STATEFILE"
"

#make script executable
chmod +x /usr/local/bin/ups-monitor.sh

#making another alert for sys resources, same methodology and add to cron set to minute checks resource-alerts.sh will point to send-telegram script
"
                                                                  
#!/bin/bash

#gets vars for resource usage
CPU=$(top -bn1 | awk '/Cpu/ {print int($2)}')
MEM=$(free | awk '/Mem:/ {printf("%.0f",$3/$2*100)}')
DISK=$(df / | awk 'END {gsub("%",""); print $5}')

#send telegram alert if cpu over 80% utilization
if [ "$CPU" -gt 80 ]; then
    /usr/local/bin/send-telegram.sh "ALERT Proxmox CPU usage is ${CPU}%"
fi

#send telegram alert if memory over 80% utilization
if [ "$MEM" -gt 80 ]; then
    /usr/local/bin/send-telegram.sh "ALERT: Proxmox RAM usage is ${MEM}%"
fi

#send telegram alert if disk storage over 80% full
if [ "$DISK" -gt 80 ]; then
    /usr/local/bin/send-telegram.sh "ALERT: Proxmox disk usage is ${DISK}%"
fi

"

#---------------autoshutdown script also pointing to telegram alert and add to cron
nano /usr/local/bin/ups-shutdown.sh
"

#!/bin/bash
UPS="cst1500s"

CHARGE=$(upsc $UPS battery.charge 2>/dev/null)

#conduct auto shutdown of Proxmox services once at/below 50%
if [ "$CHARGE" -le 50 ]; then
    /usr/local/bin/send-telegram.sh "ALERT: UPS battery at ${CHARGE}% - shutting down VMs and CTs now"
    qm shutdown --all
    pct shutdown --all
fi

#conduct auto shutdown of Proxmox host once at/below 30%
if [ "$CHARGE" -le 30 ]; then
    /usr/local/bin/telegram-send.sh "ALERT: UPS battery at ${CHARGE}% - shutting dow nProxmox Host in 60 seconds"

    sleep 60
    shutdown -h now
fi

"
